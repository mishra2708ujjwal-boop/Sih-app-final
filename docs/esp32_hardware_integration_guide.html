PRANA · Complete ESP32 & Hardware Integration Guide

Step-by-step developer manual for connecting custom ESP32/nRF/Arduino hardware sensors to the PRANA Flutter App.

Summary for Beginners
You do not need to rewrite the app to connect real hardware. The app already contains a full Bluetooth Low Energy (BLE) engine. You only need to configure three core files in Flutter and flash the provided C++ firmware onto your ESP32.

Step 3.2 — Parse JSON Sensor Payloads

Open lib/features/devices/sensor_parser.dart:

// The app accepts structured JSON packets sent from ESP32:
static Map<String, dynamic>? parseJsonTelemetry(String rawString) {
  try {
    final decoded = jsonDecode(rawString) as Map<String, dynamic>;
    return decoded;
  } catch (_) {
    return null; // Gracefully handles corrupted frames
  }
}

Step 3.3 — Switch Between Live BLE Mode and Simulator Mode

Open the app → Navigate to Devices Screen (lib/features/devices/device_screen.dart). Toggle the Live Hardware Mode switch. The app switches from MockSensorDataSource to BluetoothSensorDataSource and starts scanning for your ESP32.

4. Ready-to-Flash ESP32 Firmware (Arduino C++)

Flash this sketch using the Arduino IDE with the ArduinoJson library installed:

#include <BLEDevice.h>
#include <BLEServer.h>
#include <BLEUtils.h>
#include <BLE2902.h>
#include <ArduinoJson.h>

// UUIDs matching ble_constants.dart
#define SERVICE_UUID        "4fafc201-1fb5-459e-8fcc-c5c9c331914b"
#define CHARACTERISTIC_UUID "beb5483e-36e1-4688-b7f5-ea07361b26a8"
#define DEVICE_NAME         "PRANA_BAND_PRO"

BLEServer* pServer = NULL;
BLECharacteristic* pTelemetryCharacteristic = NULL;
bool deviceConnected = false;

class MyServerCallbacks: public BLEServerCallbacks {
    void onConnect(BLEServer* pServer) {
      deviceConnected = true;
      Serial.println("PRANA App Connected via BLE!");
    }
    void onDisconnect(BLEServer* pServer) {
      deviceConnected = false;
      Serial.println("App Disconnected! Restarting Advertising...");
      BLEDevice::startAdvertising();
    }
};

void setup() {
  Serial.begin(115200);

  // Initialize BLE Server
  BLEDevice::init(DEVICE_NAME);
  pServer = BLEDevice::createServer();
  pServer->setCallbacks(new MyServerCallbacks());

  // Create GATT Service & Characteristic
  BLEService *pService = pServer->createService(SERVICE_UUID);
  pTelemetryCharacteristic = pService->createCharacteristic(
                      CHARACTERISTIC_UUID,
                      BLECharacteristic::PROPERTY_READ   |
                      BLECharacteristic::PROPERTY_NOTIFY
                    );
  pTelemetryCharacteristic->addDescriptor(new BLE2902());

  pService->start();
  BLEAdvertising *pAdvertising = BLEDevice::getAdvertising();
  pAdvertising->addServiceUUID(SERVICE_UUID);
  pAdvertising->setScanResponse(true);
  BLEDevice::startAdvertising();

  Serial.println("ESP32 BLE Advertising as: " DEVICE_NAME);
}

void loop() {
  if (deviceConnected) {
    // 1. Read your physical sensors (MAX30102, MPU6050, DHT22)
    float hr = 76.0 + random(-2, 3);
    float spo2 = 98.2 + (random(0, 10) / 10.0);
    float temp = 36.7;
    int steps = 3420;
    int battery = 88;

    // 2. Build JSON telemetry payload
    StaticJsonDocument<256> doc;
    doc["heartRate"] = hr;
    doc["spo2"] = spo2;
    doc["temperature"] = temp;
    doc["activityLevel"] = "Moderate";
    doc["steps"] = steps;
    doc["accX"] = 0.02;
    doc["accY"] = 0.04;
    doc["accZ"] = 0.98;
    doc["battery"] = battery;

    char buffer[256];
    serializeJson(doc, buffer);

    // 3. Send over BLE Notify
    pTelemetryCharacteristic->setValue((uint8_t*)buffer, strlen(buffer));
    pTelemetryCharacteristic->notify();

    Serial.print("Sent Telemetry: ");
    Serial.println(buffer);
  }
  delay(1000); // 1-second transmission interval
}

5. Verification & Troubleshooting Checklist

  1. Bluetooth & GPS Location: Ensure both Bluetooth and Location are switched ON on your testing smartphone (Android requires location permission to discover BLE peripherals).
  2. ESP32 Serial Monitor: Open Arduino Serial Monitor at 115200 baud to confirm the device starts advertising.
  3. Pairing in App: Open the PRANA app → Devices Tab → Toggle off "Simulator Mode" → Tap "Scan for Devices" → Tap "Connect" on PRANA_BAND_PRO.
  4. Data Confirmation: The heart rate and SpO2 readings on the Home Dashboard will update in real time every second.
